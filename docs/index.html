      JavaScript (Cloudflare Worker Update Endpoint)
      
export default {
  async fetch(request, env) {
    const url = new URL(request.url);
    const slug = url.searchParams.get('slug') || 'wp-cloudflare-d1';
    return Response.json({
      slug,
      new_version: '1.4.0',
      url: `https://github.com/bhubbard/${slug}`,
      package: `https://github.com/bhubbard/${slug}/releases/download/v1.4.0/${slug}.zip`,
      tested: '6.8',
      requires_php: '8.1'
    }, { headers: { 'Cache-Control': 'public, max-age=300' } });
  }
};
